export async function executeQuickRestyle(ctx: ActionExecutionContext) {
  const contract = getActionContract("quick_restyle");
  const parsed = contract?.inputSchema.safeParse(ctx.inputSnapshot);
  if (!parsed?.success) throw new AssistantActionExecutionError("Invalid inputs", "execution_failed");

  // 1. Charge credits (idempotent)
  const creditError = await spendCreditsOrApiError({
    workspaceId: ctx.workspaceId,
    action: "restyling",
    amount: 5,
    idempotencyKey: `assistant-action:${ctx.actionId}:quick_restyle`,
    metadata: { actionId: ctx.actionId, campaignId, mode: "restyling" },
    userId: ctx.userId,
  });
  if (creditError) throw new AssistantActionExecutionError("Insufficient credits", "credit_blocked");

  // 2. Create derivation (queued)
  const derivation = await createDerivation({ campaignId, workspaceId, status: "queued", ... });

  // 3. Enqueue Inngest job
  await inngest.send({
    name: "derivation.generate",
    data: { derivationId: derivation.id, campaignId, workspaceId, triggeredByUserId: ctx.userId, locale: ctx.locale, assistantActionId: ctx.actionId, ... },
  });

  // 4. Return async mode
  return { mode: "async" as const, jobRef: { kind: "derivation" as const, id: derivation.id }, resultSummary: "..." };
}

Handler flow:
1. Parse input with `getActionContract("revise_creative")?.inputSchema.safeParse(ctx.inputSnapshot)` — throw execution_failed if invalid
2. Extract `campaignId` from the thread: `const thread = await getAssistantThreadById(ctx.workspaceId, ctx.threadId)` — need campaignId from thread
3. Determine retry attempt: read the action's retry counter from the action record (e.g., `action.attempt` or `action.retryCount`, defaulting to 0 if absent). The first attempt uses index `0`; subsequent retries use `1`, `2`, etc. NEVER use `Date.now()` — wall-clock keys are non-deterministic and break idempotency on retries that happen close together (or after process restarts). Build the retry idempotency key deterministically:
   - First attempt: `assistant-action:${ctx.actionId}:creative_revision:retry:0`
   - Retry N: `assistant-action:${ctx.actionId}:creative_revision:retry:${attemptIndex}`
   where `attemptIndex` comes from the persisted action record, not from runtime.

   Within a single attempt, `spendCreditsOrApiError` with the same key returns `duplicate` (no double charge). Across attempts, the key changes (deterministic per `attemptIndex`), so the refund-then-retry sequence always re-charges. The action is transitioned to `running` by `execute.ts` before the handler runs; idempotency at the derivation layer is enforced by checking if a derivation already exists for this action in `queued`/`processing` status (early-return with no-op) vs `failed` status (create new derivation).

4. Charge credits: `spendCreditsOrApiError({ workspaceId: ctx.workspaceId, action: "image_derivation", amount: 5, idempotencyKey: retryKey, metadata: { actionId: ctx.actionId, campaignId, mode: "creative_revision" }, userId: ctx.userId })`
5. On credit error: throw `AssistantActionExecutionError("Créditos insuficientes para esta revisão.", "credit_blocked")`
6. Confirm proposal: `await confirmCreativeRevision({ scope: { workspaceId, clientProfileId: ctx.clientProfileId, campaignId, threadId: ctx.threadId }, proposalId: parsed.data.proposalId, lineageId: parsed.data.lineageId, sourceVersionId: parsed.data.sourceVersionId, payloadDigest: parsed.data.payloadDigest, actionId: ctx.actionId, lineageHeadRevision: parsed.data.lineageHeadRevision })`
   - If idempotent return (already confirmed), skip derivation creation if one already exists
7. Create derivation: `createDerivation({ campaignId, workspaceId: ctx.workspaceId, status: "queued", generationMode: "creative_revision", variantIndex: 0, format: parsed.data.format ?? "1:1" })` — format comes from the proposal payload, but since the handler only has inputSnapshot, pass format from inputSnapshot if available, else "1:1"
8. Enqueue: `inngest.send({ name: "derivation.generate", data: { derivationId: derivation.id, campaignId, workspaceId: ctx.workspaceId, triggeredByUserId: ctx.userId, locale: ctx.locale, generationMode: "creative_revision", variantIndex: 0, format: derivation.format, assistantActionId: ctx.actionId, planVersionId: parsed.data.planVersionId } })`

9. Return: `{ mode: "async" as const, jobRef: { kind: "derivation" as const, id: derivation.id }, resultSummary: "Revisão do criativo em processamento", campaignId }`
