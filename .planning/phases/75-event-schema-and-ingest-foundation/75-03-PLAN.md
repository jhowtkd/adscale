```typescript
export async function POST(request: Request) {
  const requestId = getRequestId(request);
  try {
    const { user, workspace } = await requireWorkspaceAccess(request);
    const parsed = createBetaEventBodySchema.safeParse(await request.json());
    if (!parsed.success) return apiError("validation_error", 400, parsed.error.flatten());
    const event = await recordBetaAnalyticsEvent({
      workspaceId: workspace.id,
      userId: user.id,
      source: 'client',
      ...parsed.data,
    });
    logger.child("beta-analytics").info("beta_event.created", { eventId: event.id, workspaceId: workspace.id, userId: user.id, eventKey: event.eventKey, requestId });
    return NextResponse.json({ event }, { status: 201 });
  } catch (error) {
    if (error instanceof BetaEventPropertiesValidationError) {
      return apiError("validation_error", 400, { properties: error.message });
    }
    return handleApiError(error, "analytics.events.POST");
  }
}
```
