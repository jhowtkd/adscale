    ```typescript
    export interface AssistantTurnInput {
      workspaceId: string;
      clientProfileId: string;
      threadId: string;
      userId: string;
      userMessage: string;
      modelClient?: AssistantModelClient; // default createMiniMaxModelAdapter()
    }
    export type AssistantTurnEvent =
      | { type: "text_delta"; text: string }
      | { type: "tool_summary"; toolName: string; summary: string }
      | { type: "action_card"; actionRecordId: string; status: "pending" }
      | { type: "done"; assistantMessageId: string }
      | { type: "error"; message: string };
    export async function* runAssistantTurn(input: AssistantTurnInput): AsyncGenerator<AssistantTurnEvent>
    ```

    Orchestrator flow (integrates AI-01–05):
    1. `getAssistantThreadById` — abort if missing/wrong profile
    2. `createAssistantMessage` type `user` with `userMessage`
    3. `buildAssistantContext` → compose `AssistantModelRequest` (system prompt + history + tools from registry)
    4. Iterate `modelClient.stream()`:
       - `text_delta` → yield + accumulate full assistant text
       - `tool_call` → `evaluateToolCall`; on allow persist tool message + yield `tool_summary`/`action_card`; on deny yield `error` and continue or abort per discretion (log denial, skip tool)
    5. After stream done: `createAssistantMessage` type `assistant` with accumulated text (post-stream per Phase 178)
    6. Yield `done` with message id

    **AI-05:** Before persist, assert accumulated text and payloads pass `containsDeniedPersistenceKeys` and `assertNoReasoningInText`.
